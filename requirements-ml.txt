# requirements-ml.txt — additive ML layer for the Perch 2 *serving* image
# (deploy/Dockerfile.ml). Installed ON TOP of requirements-pipeline.txt, ONLY on
# the cluster's TF image. NOT installed in CI and NOT in the slim deploy/Dockerfile
# (which stays TF-free and serves the Stub classifier as the instant rollback).
#
# Perch 2 (Apache-2.0 SavedModel) requires TensorFlow >= 2.20. cluster-alex runs
# TF CPU-only (no cu130 wheels for TF/JAX), so we use tensorflow-cpu — smaller,
# no bundled CUDA libs. Verified 2026-06-19: tensorflow-cpu 2.20.0 ships a cp312
# manylinux wheel on PyPI (so the same python:3.12-slim base as the slim image
# works; no Python downgrade needed).
tensorflow-cpu==2.20.0

# kagglehub resolves the model from Kaggle Models. At deploy the SavedModel is
# already baked into the faun-models volume (PERCH_V2_MODEL_PATH=/models/perch2),
# so kagglehub is NOT used at runtime — it is only a belt-and-suspenders fallback
# for a self-downloading run (which would also need Kaggle creds).
kagglehub==1.0.2
